## 1. Most used commands

<CheatSheet
  rows={[
    {
      command: 'git status',
does: 'Displays the status of the Working Tree, Staging Area, and modified files',
      when: 'Before each commit or after modifications',
      example: 'git status',
    },
    {
      command: 'git status -s',
      does: 'Displays the status of files in a compressed, compact format',
      when: 'When editing a large number of files and preferring a quick list',
      example: 'git status -s',
    },
    {
      command: 'git diff',
      does: 'Displays unprocessed changes (between Working Tree and Staging Area)',
      when: 'to review modifications before git add',
      example: 'git diff',
    },
    {
command: 'git diff --cached',
      does: 'Only displays ready-to-commit changes in the Staging Area ready to commit',
      when: 'to review what will actually go into the next commit',
      example: 'git diff --cached',
    },
    {
      id: 'git-add-file',
      command: 'git add <file>',
      does: 'Prepare a selected file to the Staging Area',
      when: 'to prepare a specific file precisely and avoid making all the changes at once',
      example: 'git add src/index.ts',
    },
    {
      id: 'git-add-all',
      command: 'git add .',
      does: 'Prepare all changes to the current path and subfolders',
      when: 'to process all review edits at once',
example: 'git add .',
    },
    {
      id: 'git-add-patch',
      command: 'git add -p',
      does: 'An interactive installation that allows reviewing and selecting specific parts (Hunks) of the file',
      when: 'When there are multiple edits in a file and you want to split them over multiple commits',
      example: 'git add -p src/app.js',
    },
    {
      command: 'git commit -m "msg"',
      does: 'Records a new Snapshot of changes equipped with a clear message',
      when: 'After completion of a standalone logical feature or modification',
      example: 'git commit -m "feat: add user authentication"',
    },
  ]}
/>
