## 1. Viewing History and Diffs

<CheatSheet
  rows={[
{
      command: 'git log',
      does: 'Displays detailed commit history with the author name, date, and message',
      when: 'to review the full history of the current branch',
      example: 'git log',
    },
    {
      command: 'git log --oneline -10',
      does: 'One-line summary history for each commit of the last 10 records',
      when: 'to take a quick look at your recent edit history',
      example: 'git log --oneline -10',
    },
    {
      command: 'git log --oneline --graph --decorate --all',
      does: 'Displays a visual tree showing the branches and merging of all branches',

      example: 'git log --oneline --graph --decorate --all -15',
    },
    {
      command: 'git show <commit>',
      does: 'Checks details, content and edits (diff) for a specific commit or tag',
      when: 'to see exactly what was changed at a version or point in time',
      example: 'git show v1.0.0',
    },
    {
      command: 'git diff HEAD',
      does: 'Compares the entire Working Tree with the last commit in HEAD',
      when: 'to see all remaining changes compared to the last commit',
      example: 'git diff HEAD',
    },
{
      command: 'git diff --check',
      does: 'Detects whitespace and trailing line errors before committing',
      when: 'to avoid unintentional formatting errors in reviews',
      example: 'git diff --check',
    },
    {
      command: 'git blame <file>',
      does: 'identifies the last commit and person who changed each line in the file',
      when: 'to know the context and date of writing a specific line of code',
      example: 'git blame src/server.ts',
