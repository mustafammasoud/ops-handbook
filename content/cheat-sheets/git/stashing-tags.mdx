## 1. Stashing Commands

<CheatSheet
  rows={[
    {
      command: 'git stash',
      does: 'Causes unprocessed changes in memory and returns a clean Working Tree',
      when: 'When you need to move to another branch sooner before completing the commit',
      example: 'git stash',
    },
    {
      command: 'git stash save "work in progress"',
      does: 'Save changes with an explanatory message for easy remembering later',
      when: 'When there are multiple stashes and you want to distinguish each one',
example: 'git stash save "refactor auth logic"',
    },
    {
      command: 'git stash list',
      does: 'Displays a list of all cached edits and their handles',
      when: 'To review what you stored previously',
      example: 'git stash list',
    },
    {
      command: 'git stash pop',
      does: 'Retrieves the last saved stash and applies it to the branch and deletes it from the list',
      when: 'When you return to your branch and wish to resume your previous work',
      example: 'git stash pop',
    },
    {
      command: 'git stash apply',
      does: 'Applies the last stash to files while keeping a copy of it in the list without deleting it',
      when: 'To apply the same modifications to more than one branch of the experiment',
      example: 'git stash apply',
    },
    {
      command: 'git stash drop stash@{0}',
      does: 'Manually deletes a selected stash from the list without applying it',
      when: 'to get rid of demo work you no longer need',
      example: 'git stash drop stash@{0}',
},
  ]}
/>

## 2. Tags and Releases

<CheatSheet
  rows={[
    {
      command: 'git tag v1.0.0',
      does: 'Creates a Lightweight Tag at the current HEAD location',
      when: 'to reference an internally defined version flag',
      example: 'git tag v1.0.0',
    },
    {
      command: 'git tag -a v1.0.0 -m "Release version 1.0.0"',
      does: 'Creates an Annotated Tag that includes a message, name and date of the owner of the tag',
      when: 'For official production releases',
      example: 'git tag -a v1.0.0 -m "Release v1.0.0"',
    },
    {
      command: 'git tag -s v1.2.0 -m "Signed release"',
      does: 'Creates a cryptographically signed tag using a GPG key to ensure identity',
when: 'in high security open source and production projects',
      example: 'git tag -s v1.2.0 -m "Signed release"',
    },
    {
      command: 'git tag --list',
      does: 'displays all tags saved in the repository',
      when: 'to search for previous versions',
      example: 'git tag --list "v1.*"',
    },
    {
      command: 'git push origin v1.0.0',
      does: 'Uploads a specific tag to the remote repository',
      when: 'to publish a new version tag to GitHub',
      example: 'git push origin v1.0.0',
    },
    {
      command: 'git push origin --tags',
      does: 'uploads all new local tags to the remote repository at once',
      when: 'When you have created multiple tags and want to publish them all',
      example: 'git push origin --tags',
    },
  ]}
/>
