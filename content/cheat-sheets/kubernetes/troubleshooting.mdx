## 1. Node Maintenance & Diagnostics

<Callout type="warning" title="Drain">
The `drain` command expels Pods from the node. Always have appropriate PodDisruptionBudgets available to avoid outages in production.
</Callout>

<CheatSheet
rows={[
    {
      command: 'kubectl get nodes -o wide',
      does: 'Displays the status of all cluster nodes with OS versions, IP addresses and Kubelet version',
      when: 'To verify the readiness of the nodes (Ready / NotReady) and the correctness of their connection to the Control Plane',
      example: 'kubectl get nodes -o wide',
    },
    {
      command: 'kubectl describe node <node-name>',
      does: 'Displays comprehensive details about the node including resource constraints (Allocatable), tags, taints and pressure slices (DiskPressure/MemoryPressure)',
      when: 'When new pods fail to be scheduled on the node or it switches to NotReady',
      example: 'kubectl describe node worker-node-01',
    },
    {
      command: 'kubectl get events -A --sort-by=".metadata.creationTimestamp"',
      does: 'Fetches all block events sorted in ascending order by time of occurrence from newest to oldest',
      when: 'To track obscure problems in the cluster, such as allocation errors or failure of Liveness/Readiness health sensors',
      example: 'kubectl get events -A --sort-by=".metadata.creationTimestamp"',
},
    {
      command: 'kubectl top nodes',
      does: 'Displays the actual CPU and memory consumption of each node in the cluster',
      when: 'To evaluate resource pressure and identify nodes with high consumption (requires Metrics Server)',
      example: 'kubectl top nodes',
    },
    {
      command: 'kubectl top pods -A --sort-by=cpu',
      does: 'Displays the most processor intensive pods across all namespaces',
      when: 'to detect services with abnormal consumption and avoid exceeding the limits of available resources',
      example: 'kubectl top pods -A --sort-by=cpu',
    },
    {
      command: 'kubectl cordon <node-name>',
      does: 'Marks the node as SchedulingDisabled to prevent any new pods from being assigned to it',
      when: 'As a first step before performing maintenance or upgrade work on the server',
      example: 'kubectl cordon worker-node-02',
},
    {
      command: 'kubectl drain <node-name> --ignore-daemonsets --delete-emptydir-data',
      does: 'Evicts all removable pods from the node and safely reschedules them on other nodes',
      when: 'to completely unload the server before rebooting or upgrading the kernel and operating system',
      example: 'kubectl drain worker-node-02 --ignore-daemonsets --delete-emptydir-data',
    },
    {
      command: 'kubectl uncordon <node-name>',
      does: 'Removes the blocking state and restores the node to be ready to receive and schedule new pods',
      when: 'After completing the maintenance work and confirming the stability of the server and its return to work',
      example: 'kubectl uncordon worker-node-02',
    },
  ]}
/>
